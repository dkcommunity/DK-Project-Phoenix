# ---------------------------------------------------------------------------
# vcpkg runtime DLL'lerini GameData/'a kopyala — Client.exe'nin çalışması için
# libjpeg-turbo (jpeg62), xerces-c, zlib, lzo, ogg, vorbis gibi DLL'ler
# GameData/ altında bulunmalı. VCPKG_APPLOCAL_DEPS zaten configure sırasında
# lib'lerin yerini tespit ediyor ama runtime DLL kopyalaması için açık bir
# post-build step gerekli.
# ---------------------------------------------------------------------------
set(PHOENIX_VCPKG_BIN_DIR "${CMAKE_BINARY_DIR}/vcpkg_installed/x86-windows/bin")
if(EXISTS "${PHOENIX_VCPKG_BIN_DIR}")
    file(GLOB PHOENIX_VCPKG_DLLS "${PHOENIX_VCPKG_BIN_DIR}/*.dll")
    foreach(_dll IN LISTS PHOENIX_VCPKG_DLLS)
        get_filename_component(_dll_name "${_dll}" NAME)
        add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "${_dll}"
                "${PHOENIX_RUNTIME_DIR}/${_dll_name}"
            COMMENT "Phoenix: vcpkg DLL -> GameData/${_dll_name}"
        )
    endforeach()
    message(STATUS "Phoenix: ${PHOENIX_VCPKG_BIN_DIR} icinden DLL'ler GameData/'a kopyalanacak")
else()
    message(WARNING "vcpkg_installed bin dizini henuz yok: ${PHOENIX_VCPKG_BIN_DIR}")
endif()

# canli cozunurluk degisiminin decode/realloc dalgasi bad_alloc uretti

    # Suppress warnings: C4566 (encoding), C4091 (ignored declaration), C4996 (deprecated/unsafe)
